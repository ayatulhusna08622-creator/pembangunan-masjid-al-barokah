function hitungKeuangan(dataDonasi, dataPengeluaran) { let totalCash = 0; let totalMaterial = 0; // Hitung pemasukan donasi dataDonasi.forEach((item) => { const nominal = Number(item.nominal) || 0; const jenis = String(item.jenis || "").trim().toLowerCase(); if (jenis === "cash" || jenis === "transfer online") { totalCash += nominal; } else { totalMaterial += nominal; } }); const totalPemasukan = totalCash + totalMaterial; // Hitung total pengeluaran const totalPengeluaran = dataPengeluaran.reduce((acc, item) => { const jumlah = Number(item.jumlahHarga) || 0; return acc + jumlah; }, 0); // Hitung selisih (negatif berarti defisit, positif berarti surplus) const selisih = totalPemasukan - totalPengeluaran; return { totalCash, totalMaterial, totalPemasukan, totalPengeluaran, selisih, status: selisih < 0 ? "Defisit" : "Surplus" }; }
